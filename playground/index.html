import { useEffect, useRef } from 'react';
import { spinner } from 'lumaspinner';
// No CSS import needed! Auto-injects with strict CSP compliance.

export function UserProfile({ loading }) {
  const containerRef = useRef(null);

  useEffect(() => {
    if (loading && containerRef.current) {
      spinner.start({
        id: 'profile-loader',
        scope: 'container',
        container: containerRef.current,
        type: 'skeleton',
        skeleton: { variant: 'profile', shimmer: true },
      });
    } else {
      spinner.stop('profile-loader');
    }
    return () => spinner.stop('profile-loader');
  }, [loading]);

  return <div ref={containerRef}>...</div>;
}

import { spinner } from 'lumaspinner';

@Component({ ... })

  @ViewChild('panel') panel!: ElementRef;

    spinner.start({
      id: 'angular-dashboard',
      scope: 'container',
      container: this.panel.nativeElement,
      type: 'skeleton',
      skeleton: { variant: 'table', rows: 5 }
    });
  }

  ngOnDestroy() {
    spinner.destroy('angular-dashboard');
  }
}
